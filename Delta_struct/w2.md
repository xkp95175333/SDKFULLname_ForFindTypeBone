ผมสรุปเส้นทางที่ต้องอ่านให้ตรงกับที่คุณกำลังทำ (driver.read + actorArray) ให้เลย

1️⃣ โครงสร้างหลักของ UE ที่ใช้หา Player

ลำดับที่ใช้จริง

UWorld
 ├── PersistentLevel
 │     └── ActorArray
 │           └── Actor*
 │
 └── GameInstance
       └── LocalPlayers
             └── PlayerController
                   └── Pawn / Character
                         └── RootComponent

uintptr_t uworld = setupimguidebug::Base_PtrUworld;

uintptr_t GameInstance =
driver->read<uintptr_t>(uworld + 0x1C0);

driver->read<uintptr_t>(GameInstance + 0x38);

uintptr_t LocalPlayer =
driver->read<uintptr_t>(LocalPlayers);

uintptr_t PlayerController =
driver->read<uintptr_t>(LocalPlayer + 0x30);

5️⃣ Actor Array (ผู้เล่นทั้งหมด)

UWorld

uintptr_t ActorArray =

int ActorCount =
driver->read<int>(PersistentLevel + 0xA0);

6️⃣ loop player

    driver->read<uintptr_t>(ActorArray + i * 8);

    if (!actor)
        continue;
}

8️⃣ Bone Array (ที่คุณถาม)

UE มี 2 แบบจริง

0x6F8
0x708

ดังนั้นต้องเลือกแบบนี้

uintptr_t BoneArray =
driver->read<uintptr_t>(Mesh + 0x6F8);

if (!BoneArray)
    BoneArray =
    driver->read<uintptr_t>(Mesh + 0x708);

หรือแบบย่อ

uintptr_t BoneArray =
driver->read<uintptr_t>(Mesh + 0x6F8);

BoneArray = (!BoneArray)
 ? driver->read<uintptr_t>(Mesh + 0x708)
 : BoneArray;


---

uintptr_t worlds[2] =
{
    setupimguidebug::Base_PtrUworld,
    setupimguidebug::Base_PtrUworld2

for (int w = 0; w < 2; w++)
{
    uintptr_t world = worlds[w];

    uintptr_t level =
    driver->read<uintptr_t>(world + 0x30);

    uintptr_t actors =
    driver->read<uintptr_t>(level + 0x98);

    int count =
    driver->read<int>(level + 0xA0);

    for (int i = 0; i < count; i++)
    {
        uintptr_t actor =
        driver->read<uintptr_t>(actors + i * 8);
    }
}

12️⃣ สรุป offset สำคัญ

โครงสร้าง	offset

UWorld → PersistentLevel	0x30
Level → ActorArray	0x98
Level → ActorCount	0xA0
Actor → RootComponent	0x180
Actor → Mesh	0x3D0
Mesh → BoneArray	0x6F8 / 0x708
