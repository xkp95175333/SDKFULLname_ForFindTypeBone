```cpp

#pragma once
#include <stdint.h>

namespace UE
{

struct FVector
{
    float x;
    float y;
    float z;
};

struct FTransform
{
    float rot[4];
    float translation[3];
    float pad;
    float scale[3];
};

struct WorldPair
{
    uint64_t worldA;
    uint64_t worldB;
};

class Reader
{
public:

    void* driver;

    uint64_t worldA = 0;
    uint64_t worldB = 0;

    Reader(void* drv)
    {
        driver = drv;
    }

    template<typename T>
    T read(uint64_t addr)
    {
        return ((decltype(&::driver.read<T>))driver)->read<T>(addr);
    }

    void SetWorlds(uint64_t w1, uint64_t w2)
    {
        worldA = w1;
        worldB = w2;
    }

    uint64_t GetPersistentLevel(uint64_t world)
    {
        return read<uint64_t>(world + 0x30);
    }

    uint64_t GetGameInstance(uint64_t world)
    {
        return read<uint64_t>(world + 0x1C0);
    }

    uint64_t GetLocalPlayers(uint64_t gi)
    {
        return read<uint64_t>(gi + 0x38);
    }

    uint64_t GetLocalPlayer(uint64_t lp)
    {
        return read<uint64_t>(lp);
    }

    uint64_t GetPlayerController(uint64_t localplayer)
    {
        return read<uint64_t>(localplayer + 0x30);
    }

    uint64_t GetPawn(uint64_t controller)
    {
        return read<uint64_t>(controller + 0x350);
    }

    uint64_t GetRootComponent(uint64_t actor)
    {
        return read<uint64_t>(actor + 0x180);
    }

    FVector GetActorLocation(uint64_t actor)
    {
        uint64_t root = GetRootComponent(actor);
        return read<FVector>(root + 0x138);
    }

    uint64_t GetMesh(uint64_t actor)
    {
        return read<uint64_t>(actor + 0x3D0);
    }

    uint64_t GetBoneArray(uint64_t mesh)
    {
        uint64_t arr = read<uint64_t>(mesh + 0x6F8);

        if (!arr)
            arr = read<uint64_t>(mesh + 0x708);

        return arr;
    }

    FTransform GetBone(uint64_t mesh, int id)
    {
        uint64_t arr = GetBoneArray(mesh);

        return read<FTransform>(arr + id * 0x30);
    }

    uint64_t GetActorArray(uint64_t level)
    {
        return read<uint64_t>(level + 0x98);
    }

    int GetActorCount(uint64_t level)
    {
        return read<int>(level + 0xA0);
    }

    void IterateActors(uint64_t world)
    {
        uint64_t level = GetPersistentLevel(world);

        uint64_t actors = GetActorArray(level);

        int count = GetActorCount(level);

        for (int i = 0; i < count; i++)
        {
            uint64_t actor =
                read<uint64_t>(actors + i * 8);

            if (!actor)
                continue;

            uint64_t mesh = GetMesh(actor);

            if (!mesh)
                continue;

            FVector pos = GetActorLocation(actor);
        }
    }

    void IterateBothWorlds()
    {
        if (worldA)
            IterateActors(worldA);

        if (worldB)
            IterateActors(worldB);
    }

    uint64_t GetLocalPawn()
    {
        uint64_t gi = GetGameInstance(worldA);

        uint64_t lp = GetLocalPlayers(gi);

        uint64_t localplayer = GetLocalPlayer(lp);

        uint64_t controller = GetPlayerController(localplayer);

        return GetPawn(controller);
    }

};

}
```
